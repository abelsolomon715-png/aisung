async function triggerGeneration() { let selectedChips = document.querySelectorAll('.instrument-chip.selected'); let status = document.getElementById('status-msg'); if (selectedChips.length === 0) { status.innerText = "⚠️ Please select at least one instrument or genre element!"; status.style.color = "#ec4899"; return; } let names = []; selectedChips.forEach(c => names.push(c.innerText)); let customPrompt = "A professional studio track featuring: " + names.join(', '); status.innerText = "Submitting track to Apiframe Neural Engine..."; status.style.color = "#c4b5fd"; try { let response = await fetch('/api/generate-track', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ prompt: customPrompt }) }); let data = await response.json(); if (data.jobId) { status.innerText = "Job queued! Processing audio (this takes about 30 seconds)..."; pollJobStatus(data.jobId); } else { status.innerText = "Error: " + (data.error || "Failed to start generation."); status.style.color = "#ec4899"; } } catch (err) { status.innerText = "Network error connecting to generation backend."; status.style.color = "#ec4899"; } } async function pollJobStatus(jobId) { let status = document.getElementById('status-msg'); let interval = setInterval(async () => { try { let res = await fetch(`/api/job-status/${jobId}`); let data = await res.json(); if (data.status === 'COMPLETED' && data.result) { clearInterval(interval); status.innerHTML = `✅ Track generated successfully!
Download / Play Audio 🎵`; } else if (data.status === 'FAILED') { clearInterval(interval); status.innerText = "❌ Generation failed on the server model."; status.style.color = "#ec4899"; } } catch (e) { console.error(e); } }, 5000); }
