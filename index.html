async function handleCreateMix(prompt) { // 1. Send generation request const startResponse = await fetch('/api/generate-music', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ prompt: prompt }) }); const startData = await startResponse.json(); const taskId = startData.task_id || startData.id; if (!taskId) { alert("Generation failed to start"); return; } // 2. Poll every 5 seconds until audio_url is ready let isReady = false; let audioUrl = ""; while (!isReady) { await new Promise(resolve => setTimeout(resolve, 5000)); // wait 5 seconds const statusResponse = await fetch(`https://api.apiframe.ai/suno/status/${taskId}`, { headers: { 'Authorization': `Bearer ${YOUR_API_KEY}` } }); const statusData = await statusResponse.json(); if (statusData.status === 'complete' || statusData.audio_url) { isReady = true; audioUrl = statusData.audio_url || statusData.response?.audio_url; } } // 3. Set valid MP3 URL into player state setMixesHistory(prev => [{ title: prompt, audio_url: audioUrl, timestamp: new Date().toLocaleTimeString() }, ...prev]); }
